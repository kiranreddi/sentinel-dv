\section{Model Context Protocol}
\begin{frame}[fragile]{Model Context Protocol (MCP)}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{What is MCP?}
        \small
        \begin{itemize}
          \item Open standard by Anthropic (Nov 2024)
          \item JSON-RPC 2.0 over stdio or SSE
          \item AI \textbf{client} discovers and calls \textbf{tools}
                exposed by a \textbf{server}
          \item Adopted by Claude, GPT-4o, Copilot, Cursor
          \item Zero code changes to the AI — just plug in the server
        \end{itemize}
      \end{block}

      \vspace{0.3em}
      \begin{exampleblock}{MCP Tool Call (JSON-RPC 2.0)}
        \footnotesize
\begin{semiverbatim}\color{white}
\{"jsonrpc":"2.0","id":1,
 "method":"tools/call",
 "params":\{
   "name":"regression.health",
   "arguments":\{\}
 \}\}
--> \{"health_score":87,
     "status":"pass",
     "failing_tests":2\}
\end{semiverbatim}
      \end{exampleblock}

    \column{0.44\textwidth}
      \begin{block}{Why MCP for DV?}
        \small
        \begin{itemize}
          \item Sim artifacts $\rightarrow$ structured JSON tools
          \item AI reasons over \emph{your} regression
          \item Works with any LLM that supports MCP
          \item Runs in CI — no GUI required
          \item Output schemas validated (Pydantic)
          \item Path redaction built in
          \item \textbf{Tool-agnostic}: VCS, Questa, Xcelium, Verilator
        \end{itemize}
      \end{block}
      \vspace{0.3em}
      \begin{alertblock}{Token Efficiency}
        \small
        Raw logs: \textbf{100K+ tokens}/run\\
        MCP structured output: \textbf{$<$500 tokens}\\
        \textbf{200$\times$ context savings}
      \end{alertblock}
  \end{columns}
\end{frame}

\begin{frame}[fragile]{DV Intelligence — Regression Health Score}
  \begin{columns}[T]
    \column{0.52\textwidth}
      \begin{block}{Composite Health Formula}
        \[
          H = 0.30 P + 0.35 C + 0.15 A + 0.10 F + 0.10 X
        \]
        \vspace{-0.5em}
        \begin{itemize}
          \item $P$ = pass rate (30\% weight)
          \item $C$ = coverage score (35\% weight)
          \item $A$ = assertion health (if indexed)
          \item $F$ = flakiness inverse (10\% weight)
          \item $X$ = cross-sim consistency (10\% weight)
          \item Missing components are flagged and weights renormalize
        \end{itemize}
      \end{block}
      \vspace{0.3em}
      \begin{block}{Bands}
        \begin{tabular}{ll}
          $H \geq 90$ & \textcolor{sdvgreen}{\textbf{sign-off-ready}} \\
          $H \geq 75$ & \textcolor{sdvblue}{\textbf{minor-issues}} \\
          $H \geq 55$ & \textcolor{sdvorange}{\textbf{coverage-gaps}} \\
          $H < 55$    & \textcolor{sdvred}{\textbf{not-ready}} \\
        \end{tabular}
      \end{block}

    \column{0.44\textwidth}
      \begin{exampleblock}{Example Output}
\begin{semiverbatim}\color{white}

  "health_score": 77.5,

    "assertions":   100.0,
    "flakiness":    98.1,
    "cross_sim":    87.5
  },
  "recommendations": [
    "Coverage below target (51%)",
    "3 tests flaky across sims"
  ]
}
\end{semiverbatim}

\begin{frame}[fragile]{Token \& Context Efficiency}
  \begin{columns}[T]
    \column{0.50\textwidth}
      \begin{block}{The Problem with Raw Logs}
        \small
        \begin{itemize}
          \item A single VCS simulation log: \textbf{50K--500K lines}
          \item Raw logs to an LLM: context exhausted fast
          \item One regression run: $>$1M tokens of raw text
          \item Result: truncation, hallucination, high API cost
        \end{itemize}
      \end{block}
      \vspace{0.3em}
      \begin{alertblock}{Tokens Consumed Comparison}
        \footnotesize
        \begin{tabular}{lr}
          \toprule
          \textbf{Approach} & \textbf{Tokens} \\
          \midrule
          Raw UVM logs (50 tests) & $\sim$2,000,000 \\
          Raw HTML coverage report & $\sim$120,000 \\
          \textbf{MCP structured output} & \textbf{$\sim$3,000} \\
          \bottomrule
        \end{tabular}\\[2pt]
        \textbf{$>$600$\times$ context savings}
      \end{alertblock}

    \column{0.46\textwidth}
      \begin{block}{How Sentinel DV Achieves This}
        \small
        \begin{itemize}
          \item Parse once at index time (offline)
          \item Expose only \emph{structured summaries}
          \item Bounded output schemas (Pydantic)
          \item Redaction removes paths/names
        \end{itemize}
